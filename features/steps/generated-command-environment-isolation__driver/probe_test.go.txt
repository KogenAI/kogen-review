package review

import (
	"encoding/json"
	"flag"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"testing"

	"github.com/KogenAI/kogen-executor/core"
)

var generatedEnvironmentWant = flag.String("generated-environment-want", "", "expected generated outcomes in path order")

func TestAcceptanceGeneratedCommandEnvironment(t *testing.T) {
	want := strings.Split(*generatedEnvironmentWant, ",")
	if len(want) != 3 {
		t.Fatalf("probe requires three expected outcomes, got %q", want)
	}
	const emit = `emit() {
  {
    printf '# Code generated by %s; DO NOT EDIT.\n' "$1"
    i=1
    while [ "$i" -le 700 ]; do
      printf '%s row %s\n' "$2" "$i"
      i=$((i+1))
    done
  } > "$2"
}
`
	t.Setenv("KOGEN_REGEN_TEST_SENTINEL", "parent-only")
	bin := t.TempDir()
	t.Setenv("PATH", bin+string(os.PathListSeparator)+os.Getenv("PATH"))
	if err := os.WriteFile(filepath.Join(bin, "kogen-fixture-emitter"), []byte("#!/bin/sh\n"+emit+`emit "kogen-fixture-emitter $1 $2" "$1" || exit
exit "$2"
`), 0755); err != nil {
		t.Fatal(err)
	}

	r := newEdgeRepo(t)
	start := r.commit("start", nil)
	// Omit the CSVs from the scratch archive: a no-op cannot pass by leaving old bytes alone.
	r.write(".gitattributes", "*.csv export-ignore\n")
	r.write("gen/isolated.sh", emit+`[ "${KOGEN_REGEN_TEST_SENTINEL+x}" != x ] || exit 23
emit 'sh gen/isolated.sh' isolated.csv
`)
	cases := []struct{ name, path, command string }{
		{"failed_exit", "failed.csv", "kogen-fixture-emitter failed.csv 3"},
		{"sentinel_isolated", "isolated.csv", "sh gen/isolated.sh"},
		{"path_tool", "via-path.csv", "kogen-fixture-emitter via-path.csv 0"},
	}
	contents := make([]string, len(cases))
	for i, c := range cases {
		var body strings.Builder
		fmt.Fprintf(&body, "# Code generated by %s; DO NOT EDIT.\n", c.command)
		for row := 1; row <= 700; row++ {
			fmt.Fprintf(&body, "%s row %d\n", c.path, row)
		}
		contents[i] = body.String()
		r.write(c.path, contents[i])
	}
	r.write("big.txt", exactly(reviewCeiling+20000, "b"))
	r.commit("candidate", nil)

	x, _, dir, res, err := wideReview(t, r, start)
	if err != nil || res.Record.Verdict != "approve" || len(res.Record.Findings) != 0 || len(x.briefs) == 0 {
		t.Fatalf("fixture review did not approve in parts without findings: verdict %q, findings %v, brief count %d, error %v", res.Record.Verdict, res.Record.Findings, len(x.briefs), err)
	}
	materialDir := filepath.Join(dir, "executor-review")
	raw, err := os.ReadFile(filepath.Join(materialDir, "review-1-material.json"))
	if err != nil {
		t.Fatal(err)
	}
	var material Material
	if err := json.Unmarshal(raw, &material); err != nil {
		t.Fatal(err)
	}
	if len(material.Generated) != len(cases) {
		t.Fatalf("got %d generated records, want %d: %v", len(material.Generated), len(cases), material.Generated)
	}
	source, err := os.ReadFile(filepath.Join(materialDir, "review-1-source.txt"))
	if err != nil {
		t.Fatal(err)
	}
	var brief strings.Builder
	for _, b := range x.briefs {
		brief.WriteString(b.Dynamic)
	}
	for i, c := range cases {
		t.Run(c.name, func(t *testing.T) {
			content := contents[i]
			digest := core.HashHex([]byte(content))
			wantRecord := GeneratedFile{Path: c.path, Command: c.command, SHA256: digest, Outcome: want[i]}
			if got := material.Generated[i]; got != wantRecord {
				t.Errorf("%s: generated outcome %q, want %q (record %+v, want %+v)", c.path, got.Outcome, want[i], got, wantRecord)
			}
			for _, observed := range []struct{ name, text string }{{"retained source", string(source)}, {"delivered parts", brief.String()}} {
				if want[i] == "kept" {
					if !strings.Contains(observed.text, content) {
						t.Errorf("%s: %s misses the kept text", c.path, observed.name)
					}
					continue
				}
				summary := fmt.Sprintf("generated: %d bytes, SHA-256 %s; regenerated by `%s` with identical output", len(content), digest, c.command)
				if !strings.Contains(observed.text, summary) || strings.Contains(observed.text, c.path+" row ") {
					t.Errorf("%s: %s must carry its digest instead of its rows", c.path, observed.name)
				}
			}
		})
	}
}
